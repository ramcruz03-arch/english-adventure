name: build

# PLACEMENT MATTERS: this file must sit at the REPOSITORY ROOT as
# .github/workflows/build.yml. GitHub only reads workflows from the root -
# a copy inside artifacts/english-adventure/.github/ is never executed.
#
# The app itself lives in a subdirectory, so every step runs from there via
# the defaults block below. If you ever move the Flutter project to the repo
# root, change working-directory to '.' and drop the prefix on the artifact
# path at the bottom.

on:
  push:
    branches: [main]
  pull_request:
  workflow_dispatch:

defaults:
  run:
    working-directory: artifacts/english-adventure

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4

      # AGP with compileSdk 35 needs JDK 17. The Kotlin jvmTarget stays at 11
      # via build.gradle.kts; this is the toolchain, not the target.
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true

      - run: flutter --version

      - name: Resolve dependencies
        run: flutter pub get

      # Fails the run on any analyzer error. Warnings and lints from
      # flutter_lints do not fail it - drop --no-fatal-infos to tighten that.
      - name: Analyze
        run: flutter analyze --no-fatal-infos

      - name: Test
        run: flutter test

      # A hard gate, not an advisory one. The validator's own header calls it
      # the most important hundred lines in the repo: it is what guarantees a
      # child is never shown a word using a sound nobody has taught her yet.
      - name: Validate content pack
        run: dart run tools/validate_content.dart

      - name: Build APK
        run: flutter build apk --release

      # upload-artifact paths are relative to the workspace root, NOT to the
      # defaults working-directory above - so this one keeps the full prefix.
      - uses: actions/upload-artifact@v4
        with:
          name: english-adventure-apk
          path: artifacts/english-adventure/build/app/outputs/flutter-apk/app-release.apk
          if-no-files-found: error
          retention-days: 14
